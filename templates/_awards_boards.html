{# Macros for the 2026+ team-format season leaderboards on /awards. Every board is rendered from the same row shape (see routes/pages/awards_boards.build_boards), so one set of macros serves both grains: anglers and exact boat pairings. #} {# Gold / silver / bronze rank pill, matching the badge colors already used by the standings tables on this page. `tied` prefixes "T-" so rows sharing a rank don't imply an order the data doesn't support. #} {% macro rank_badge(rank, tied=False) %} {%- if rank == 1 %}{% if tied %}T-{% endif %}1 {%- elif rank == 2 %}{% if tied %}T-{% endif %}2 {%- elif rank == 3 %}{% if tied %}T-{% endif %}3 {%- else %}{% if tied %}T-{% endif %}{{ rank }}{% endif %} {%- endmacro %} {# Pluralizes a count's noun: "1 event", "2 events". Irregular plurals pass their own form -- "fish" and "finishes" don't take a bare trailing s. #} {% macro plural(count, word, plural_word='') %}{{ count }} {{ word if count == 1 else (plural_word or word ~ 's') }}{% endmacro %} {# Secondary line under a name on a board card.
